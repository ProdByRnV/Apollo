target_compile_definitions(ApolloTests PRIVATE
    JUCE_STANDALONE_APPLICATION=1
    # Keep the test binary free of optional platform dependencies: the suite
    # needs no browser, and requiring one would make the Linux CI image heavier
    # than the tests justify.
    JUCE_WEB_BROWSER=0
    # Headless: the bridge is exercised through its protocol layer, which needs
    # no browser. The processor builds its generic-editor fallback here.
    APOLLO_WITH_WEBVIEW=0
    # JUCE's *own* internal unit tests are not part of Apollo's suite.
    JUCE_UNIT_TESTS=0)

# apollo::strict_warnings is deliberately not applied here: this target compiles
# JUCE module sources, which are held to JUCE's own warning standard rather than
# Apollo's. Apollo code under test lives in apollo::core, which is strict.
#
# apollo::engine is linked so that the processor is tested as the plugin
# actually builds it, from the same sources.
# juce_dsp is linked for its FFT, which the spectral tests use to measure
# aliasing. It is a JUCE module with no external dependency, and the DSP engine
# itself does not use it Ã¢â‚¬â€ the analysis lives only in the tests.
target_link_libraries(ApolloTests PRIVATE
    apollo::engine
    juce::juce_dsp
    apollo::project_options
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags)

# The test tree is on the include path so test-only headers (the benchmarks)
# can be included by path, the same way Apollo headers are.
target_include_directories(ApolloTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

set_target_properties(ApolloTests PROPERTIES FOLDER "Apollo/Tests")

add_test(NAME apollo.unit COMMAND ApolloTests)

# The timeout is a deadlock guard, not a performance budget.
#
# 300 seconds was ample when the suite was small and every job ran it
# optimised. It stopped being ample in Phase 4b: the sanitizer job builds Debug
# with ASan and UBSan under Clang, and the source-section tests render a great
# deal of audio Ã¢â‚¬â€ 32 voices with 16-voice unison on both oscillators, spectral
# sweeps, and per-sample gain ramps. That job reached 92% of the suite and was
# killed at exactly 300.02 s, which reported as a failure while nothing had
# actually gone wrong.
#
# Measured on the development machine, for scale: 24 s optimised, 82 s in
# Debug. The sanitized Linux build is roughly four times the Debug figure again.
# 1800 s leaves room for the phases still to come without turning a slow
# configuration into a red build, and still catches a genuine hang far inside
# GitHub's six-hour job limit.
#
# This deliberately does not police how long the suite takes. Phase 4c measures
# CPU cost properly, and a real performance budget belongs there rather than
# hidden in a CI timeout.
set_tests_properties(apollo.unit PROPERTIES
    TIMEOUT 1800
    LABELS "unit")

# The regression renders again, on their own, so that their measurements reach
# the log of a run that PASSED.
#
# ctest shows a test's output only when it fails, which is right for a suite of
# two million assertions and wrong for the handful of lines that say how far
# each render sat from its reference on this platform. That figure is the whole
# cross-platform evidence for whether the tolerances are set sensibly, and it is
# worthless if it is only visible when something is already broken.
#
# Labelled separately so the ordinary run does not pick it up: CI runs the suite
# with `-L unit` and names this one explicitly with `-V`.
add_test(NAME apollo.regression COMMAND ApolloTests --category Regression)

set_tests_properties(apollo.regression PROPERTIES
    TIMEOUT 900
    LABELS "report")

# The host harness (Phase 10e, ADR-0075).
#
# A second test binary, and deliberately not part of ApolloTests: that suite
# links Apollo's engine and drives the processor through its C++ interface,
# which no host ever does. This one links no Apollo code at all. It loads the
# VST3 bundle the build just produced, through JUCE's VST3 hosting
# implementation, and talks to it through the VST3 interfaces a DAW uses.
#
# juce_audio_processors_headless rather than juce_audio_processors: hosting
# needs no windowing, and the headless module keeps GTK and X11 out of a binary
# that has to run on a CI runner with no display.
#
# What it reads from Apollo's source tree is the parameter registry, which is
# plain data with no JUCE dependency — the same list a host is shown, used to
# check that the host was shown it correctly.
if(TARGET Apollo_VST3)
    juce_add_console_app(ApolloHostTests
        PRODUCT_NAME "Apollo Host Tests")

    target_sources(ApolloHostTests PRIVATE
        Host/HostMain.cpp
        Host/HostedApollo.cpp
        Host/HostedApollo.h
        Host/ParameterIdentities.h
        Host/ParameterIdentityGenerator.cpp
        # Generated by `ApolloHostTests --parameter-ids`, and checked in.
        Host/ParameterIdentities.cpp
        Host/DiscoveryTests.cpp
        Host/StateTests.cpp
        Host/ProcessingTests.cpp
        Host/BypassTests.cpp
        Host/PackageTests.cpp
        Host/ReliabilityTests.cpp
        Package/BinaryImage.cpp
        Package/BinaryImage.h
        # footprintBytes: the reliability soak asks whether memory comes back,
        # and 10c already had a portable way to ask.
        Performance/Machine.cpp
        Performance/Machine.h)

    target_compile_definitions(ApolloHostTests PRIVATE
        JUCE_STANDALONE_APPLICATION=1
        JUCE_PLUGINHOST_VST3=1
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_UNIT_TESTS=0
        # A host pumps its message loop; so must the harness, to receive the
        # latency change a plugin announces from its message thread.
        JUCE_MODAL_LOOPS_PERMITTED=1
        # The bundle this build produced. Overridable with --plugin, to test the
        # copy a DAW actually has installed. JUCE stores the path as a generator
        # expression of its own, hence the evaluation.
        "APOLLO_HOST_TEST_DEFAULT_BUNDLE=\"$<TARGET_GENEX_EVAL:Apollo_VST3,$<TARGET_PROPERTY:Apollo_VST3,JUCE_PLUGIN_ARTEFACT_FILE>>\""
        "APOLLO_HOST_TEST_EXPECTED_VERSION=\"${PROJECT_VERSION}\"")

    # What the build asked the toolchain to produce, so the package test can
    # check the artefact against the intention rather than against a guess.
    #
    # A plain if() rather than a generator expression: the value is a
    # comma-separated list, and a comma inside $<BOOL:...> is read as the
    # expression's own argument separator — which is a configure error on the
    # one platform that sets more than one architecture.
    if(APOLLO_EXPECTED_ARCHITECTURES)
        target_compile_definitions(ApolloHostTests PRIVATE
            "APOLLO_EXPECTED_ARCHITECTURES=\"${APOLLO_EXPECTED_ARCHITECTURES}\"")
    endif()

    target_link_libraries(ApolloHostTests PRIVATE
        juce::juce_audio_processors_headless
        apollo::project_options
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)

    target_include_directories(ApolloHostTests PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}"
        "${PROJECT_SOURCE_DIR}/Source")

    # The harness is meaningless without the bundle it loads, so building it
    # builds the plugin first.
    add_dependencies(ApolloHostTests Apollo_VST3)

    set_target_properties(ApolloHostTests PROPERTIES FOLDER "Apollo/Tests")

    add_test(NAME apollo.host COMMAND ApolloHostTests)

    set_tests_properties(apollo.host PROPERTIES
        TIMEOUT 900
        LABELS "host")
endif()
